## How It Works

1. **Capture user input** — Your user types text into a form field
2. **Send to Whet Data** — Make a request to the `/api/ult` endpoint
3. **Receive intelligent feedback** — Get structured nudges and suggestions
4. **Display to users** — Show the feedback in your interface

    <CodeGroup>

    ```bash cURL
      curl -X POST https://whetdata.com/api/ult \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer YOUR_API_KEY" \
        -d '{
          "action": "nudges",
          "ruleset": "product_description",
          "text": "Modern sneakers for daily wear",
          "max_nudges": 3
        }'
    ```

    
    ```tsx TypeScript
    export async function createNudges(text: string): Promise<any> {
      const API_URL = "https://whetdata.com/api/ult";
      const API_KEY = "Reference your API KEY!";
      const ruleset = "product_description";
      const max_nudges = 3;
    
      try {
        const response = await fetch(API_URL, {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${API_KEY}`,
          },
          body: JSON.stringify({
            action: "nudges",
            ruleset,
            text,
            max_nudges,
          }),
        });
    
        if (!response.ok) {
          throw new Error(`Request failed: ${response.status}`);
        }
    
        const data = await response.json();
        return data;
      } catch (error) {
        console.error("Error creating nudges:", error);
        throw error;
      }
    }
    ```

    
    ```python Python
    import requests
    
    def create_nudges(text: str):
        API_URL = "https://whetdata.com/api/ult"
        API_KEY = "Reference your API KEY!"
    
        payload = {
            "action": "nudges",
            "ruleset": "product_description",
            "text": text,
            "max_nudges": 3
        }
    
        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {API_KEY}"
        }
    
        response = requests.post(API_URL, json=payload, headers=headers)
    
        if not response.ok:
            raise Exception(f"Request failed: {response.status_code}")
    
        return response.json()
    ```
