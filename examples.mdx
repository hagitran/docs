## Overview

This page provides practical examples of how to use WhetInput in various real-world scenarios. Each example includes working code and explains common use cases.

## Basic Examples

### Simple Text Input

The most basic usage - a smart text input with nudges:

```tsx
import { WhetInput } from '@whetdata/input';
import { useState } from 'react';

function BasicExample() {
  const [text, setText] = useState('');

  return (
    <WhetInput
      value={text}
      onChange={setText}
      placeholder="Start typing..."
      target="content"
      nudgePackId="professional_writing"
      apiKey="your-api-key"
    />
  );
}
```

### Contact Form

Complete contact form with multiple WhetInput components:

```tsx
import { WhetInput } from '@whetdata/input';
```

### Comment Section

Smart comment input with moderation:

```tsx
import { WhetInput } from '@whetdata/input';
import { useState } from 'react';

function CommentSection() {
  const [comment, setComment] = useState('');
  const [comments, setComments] = useState<string[]>([]);

  const handleSubmit = () => {
    if (comment.trim()) {
      setComments([...comments, comment]);
      setComment('');
    }
  };

  return (
    <div className="space-y-4">
      <h3 className="text-xl font-bold">Comments</h3>
      
      <div>
        <WhetInput
          value={comment}
          onChange={setComment}
          placeholder="Share your thoughts..."
          target="comment"
          nudgePackId="content_moderation"
          apiKey="your-api-key"
          multiline
        />
        <button
          onClick={handleSubmit}
          className="mt-2 px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
        >
          Post Comment
        </button>
      </div>

      <div className="space-y-2">
        {comments.map((c, i) => (
          <div key={i} className="p-4 bg-gray-100 rounded">
            {c}
          </div>
        ))}
      </div>
    </div>
  );
}
```

## Advanced Use Cases

### Multi-Language Support

Smart input that adapts to different languages:

```tsx
import { WhetInput } from '@whetdata/input';
import { useState } from 'react';

function MultiLanguageInput() {
  const [language, setLanguage] = useState('en');
  const [text, setText] = useState('');

  return (
    <div className="space-y-4">
      <div>
        <label className="block mb-2 font-medium">Language</label>
        <select
          value={language}
          onChange={(e) => setLanguage(e.target.value)}
          className="px-4 py-2 border rounded"
        >
          <option value="en">English</option>
          <option value="es">Spanish</option>
          <option value="fr">French</option>
          <option value="de">German</option>
        </select>
      </div>

      <WhetInput
        value={text}
        onChange={setText}
        placeholder={`Write in ${language}...`}
        target="content"
        nudgePackId={`writing_${language}`}
        apiKey="your-api-key"
        metadata={{ language }}
      />
    </div>
  );
}
```

### React Hook Form Integration

Integrating WhetInput with React Hook Form:

```tsx
import { WhetInput } from '@whetdata/input';
import { useForm, Controller } from 'react-hook-form';

interface FormData {
  email: string;
  message: string;
}

function HookFormExample() {
  const { control, handleSubmit, formState: { errors } } = useForm<FormData>();

  const onSubmit = (data: FormData) => {
    console.log(data);
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="space-y-4">
      <div>
        <label className="block mb-2 font-medium">Email</label>
        <Controller
          name="email"
          control={control}
          rules={{ required: 'Email is required' }}
          render={({ field }) => (
            <WhetInput
              value={field.value}
              onChange={field.onChange}
              placeholder="your@email.com"
              target="email"
              nudgePackId="email_validation"
              apiKey="your-api-key"
            />
          )}
        />
        {errors.email && (
          <span className="text-red-500 text-sm">{errors.email.message}</span>
        )}
      </div>

      <div>
        <label className="block mb-2 font-medium">Message</label>
        <Controller
          name="message"
          control={control}
          rules={{ required: 'Message is required' }}
          render={({ field }) => (
            <WhetInput
              value={field.value}
              onChange={field.onChange}
              placeholder="Your message..."
              target="content"
              nudgePackId="professional_writing"
              apiKey="your-api-key"
              multiline
            />
          )}
        />
        {errors.message && (
          <span className="text-red-500 text-sm">{errors.message.message}</span>
        )}
      </div>

      <button
        type="submit"
        className="px-6 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
      >
        Submit
      </button>
    </form>
  );
}
```

## Common Patterns

<AccordionGroup>
  <Accordion title="Loading States">
    ```tsx
    const [isLoading, setIsLoading] = useState(false);

    <WhetInput
      value={text}
      onChange={setText}
      disabled={isLoading}
      placeholder={isLoading ? "Loading..." : "Type here..."}
    />
    ```
  </Accordion>

  <Accordion title="Validation">
    ```tsx
    const [error, setError] = useState('');

    const handleChange = (value: string) => {
      setText(value);
      if (value.length < 10) {
        setError('Must be at least 10 characters');
      } else {
        setError('');
      }
    };

    <WhetInput
      value={text}
      onChange={handleChange}
      error={error}
    />
    ```
  </Accordion>

  <Accordion title="Character Counter">

    ```tsx
    const [text, setText] = useState('');

      <div className="text-sm text-gray-600 mt-1">
        {text.length}/{maxLength}
      </div>
    ```
