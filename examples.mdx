## Classification Examples

### Example 1: Validate Product Description

Check if a product description meets quality standards before publishing.

<CodeGroup>

```bash cURL
curl -X POST https://whetdata.com/api/ult \
  -H "x-api-key: whet_your_api_key_here" \
  -H "Content-Type: application/json" \
  -d '{
    "action": "classification",
    "ruleset": "product_descriptions",
    "text": "Premium wireless headphones with active noise cancellation. Features 30-hour battery life, crystal-clear audio, and comfortable over-ear design. Perfect for music lovers, travelers, and professionals.",
    "threshold": 0.8
  }'
```

```python Python
import requests

def validate_product_description(description: str) -> dict:
    url = "https://whetdata.com/api/ult"
    headers = {
        "x-api-key": "whet_your_api_key_here",
        "Content-Type": "application/json"
    }
    payload = {
        "action": "classification",
        "ruleset": "product_descriptions",
        "text": description,
        "threshold": 0.8
    }
    
    response = requests.post(url, headers=headers, json=payload)
    return response.json()

# Usage
result = validate_product_description(
    "Premium wireless headphones with active noise cancellation..."
)

if result['result']['valid']:
    print(f"✓ Description approved! Score: {result['result']['score']}")
else:
    print(f"✗ Description needs improvement. Score: {result['result']['score']}")
```

```tsx TypeScript
async function validateProductDescription(description: string) {
  const response = await fetch("https://whetdata.com/api/ult", {
    method: "POST",
    headers: {
      "x-api-key": "whet_your_api_key_here",
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      action: "classification",
      ruleset: "product_descriptions",
      text: description,
      threshold: 0.8,
    }),
  });

  const data = await response.json();
  
  if (data.result.valid) {
    console.log(`✓ Description approved! Score: ${data.result.score}`);
  } else {
    console.log(`✗ Needs improvement. Score: ${data.result.score}`);
  }
  
  return data;
}

// Usage
await validateProductDescription(
  "Premium wireless headphones with active noise cancellation..."
);
```

### Example 2: Batch Validation

Validate multiple product descriptions efficiently.

<CodeGroup>

```python Python
import requests
from typing import List, Dict

def batch_validate_descriptions(descriptions: List[str]) -> List[Dict]:
    url = "https://whetdata.com/api/ult"
    headers = {
        "x-api-key": "whet_your_api_key_here",
        "Content-Type": "application/json"
    }
    
    results = []
    for desc in descriptions:
        payload = {
            "action": "classification",
            "ruleset": "product_descriptions",
            "text": desc,
            "threshold": 0.75
        }
        response = requests.post(url, headers=headers, json=payload)
        results.append(response.json())
    
    return results

# Usage
descriptions = [
    "Wireless headphones with great sound",
    "Premium over-ear headphones featuring active noise cancellation...",
    "Headphones"
]

results = batch_validate_descriptions(descriptions)

for i, result in enumerate(results, 1):
    status = "✓ Pass" if result['result']['valid'] else "✗ Fail"
    print(f"{i}. {status} - Score: {result['result']['score']}")
```

```tsx TypeScript
interface ValidationResult {
  success: boolean;
  result: {
    valid: boolean;
    score: number;
    reason: string;
  };
}

async function batchValidateDescriptions(
  descriptions: string[]
): Promise<ValidationResult[]> {
  const url = "https://whetdata.com/api/ult";
  
  const promises = descriptions.map(async (text) => {
    const response = await fetch(url, {
      method: "POST",
      headers: {
        "x-api-key": "whet_your_api_key_here",
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        action: "classification",
        ruleset: "product_descriptions",
        text,
        threshold: 0.75,
      }),
    });
    return response.json();
  });

  return Promise.all(promises);
}

// Usage
const descriptions = [
  "Wireless headphones with great sound",
  "Premium over-ear headphones featuring active noise cancellation...",
  "Headphones"
];

const results = await batchValidateDescriptions(descriptions);
results.forEach((result, i) => {
  const status = result.result.valid ? "✓ Pass" : "✗ Fail";
  console.log(`${i + 1}. ${status} - Score: ${result.result.score}`);
});
```

</CodeGroup>

## Complete Workflow Example

### Example 7: Full Content Pipeline

Combine all three actions for a complete content quality workflow.

```python Python
import requests

class WhetDataClient:
    def __init__(self, api_key: str):
        self.api_key = api_key
        self.base_url = "https://whetdata.com/api/ult"
        self.headers = {
            "x-api-key": api_key,
            "Content-Type": "application/json"
        }
    
    def _request(self, payload: dict) -> dict:
        response = requests.post(self.base_url, headers=self.headers, json=payload)
        return response.json()
    
    def clean(self, text: str) -> str:
        result = self._request({"action": "clean", "text": text})
        return result['corrected_text']
    
    def classify(self, text: str, ruleset: str, threshold: float = 0.75) -> dict:
        return self._request({
            "action": "classification",
            "ruleset": ruleset,
            "text": text,
            "threshold": threshold
        })
    
    def get_nudges(self, text: str, ruleset: str, max_nudges: int = 5) -> list:
        result = self._request({
            "action": "nudges",
            "ruleset": ruleset,
            "text": text,
            "max_nudges": max_nudges
        })
        return result['nudges']

# Usage: Complete workflow
client = WhetDataClient("whet_your_api_key_here")

# Step 1: Fix any spelling/grammar errors
raw_text = "This product is grate for evreyday use."
cleaned_text = client.clean(raw_text)
print(f"1. Cleaned: {cleaned_text}")

# Step 2: Check if it meets quality standards
validation = client.classify(
    text=cleaned_text,
    ruleset="product_descriptions",
    threshold=0.8
)

if not validation['result']['valid']:
    print(f"2. Quality check failed (score: {validation['result']['score']})")
    
    # Step 3: Get suggestions to improve
    suggestions = client.get_nudges(
        text=cleaned_text,
        ruleset="product_descriptions",
        max_nudges=3
    )
    
    print("3. Suggestions to improve:")
    for suggestion in suggestions:
        print(f"   • {suggestion['suggestion']}")
else:
    print(f"2. ✓ Quality approved! (score: {validation['result']['score']})")
    print("3. Ready to publish!")
```

## Error Handling

### Example 8: Handle Errors Gracefully

Always implement proper error handling in production.

<CodeGroup>

```python Python
import requests
from typing import Optional

def safe_api_call(text: str, action: str) -> Optional[dict]:
    url = "https://whetdata.com/api/ult"
    headers = {
        "x-api-key": "whet_your_api_key_here",
        "Content-Type": "application/json"
    }
    
    try:
        response = requests.post(
            url, 
            headers=headers, 
            json={"action": action, "text": text},
            timeout=10
        )
        response.raise_for_status()
        return response.json()
        
    except requests.exceptions.HTTPError as e:
        if e.response.status_code == 429:
            print("⚠️  Rate limit reached. Please wait before retrying.")
        elif e.response.status_code == 401:
            print("❌ Invalid API key. Check your credentials.")
        elif e.response.status_code == 400:
            print(f"❌ Bad request: {e.response.json().get('error')}")
        else:
            print(f"❌ HTTP error: {e.response.status_code}")
        return None
        
    except requests.exceptions.Timeout:
        print("⚠️  Request timed out. Try again.")
        return None
        
    except requests.exceptions.RequestException as e:
        print(f"❌ Network error: {str(e)}")
        return None

# Usage
result = safe_api_call("Test text", "clean")
if result:
    print("Success:", result)
```

```tsx TypeScript
async function safeApiCall(
  text: string, 
  action: string
): Promise<any | null> {
  try {
    const response = await fetch("https://whetdata.com/api/ult", {
      method: "POST",
      headers: {
        "x-api-key": "whet_your_api_key_here",
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ action, text }),
      signal: AbortSignal.timeout(10000), // 10 second timeout
    });

    if (!response.ok) {
      const error = await response.json();
      
      switch (response.status) {
        case 429:
          console.error("⚠️  Rate limit reached. Please wait.");
          break;
        case 401:
          console.error("❌ Invalid API key.");
          break;
        case 400:
          console.error(`❌ Bad request: ${error.error}`);
          break;
        default:
          console.error(`❌ HTTP error: ${response.status}`);
      }
      return null;
    }

    return await response.json();
    
  } catch (error) {
    if (error.name === 'TimeoutError') {
      console.error("⚠️  Request timed out. Try again.");
    } else {
      console.error(`❌ Network error: ${error.message}`);
    }
    return null;
  }
}

// Usage
const result = await safeApiCall("Test text", "clean");
if (result) {
  console.log("Success:", result);
}
```
